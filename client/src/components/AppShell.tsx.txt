/**
 * AppShell — the authenticated application layout.
 *
 * Structure:
 *   <AppShell>          ← provides sidebar + topbar + scroll area
 *     <PageContent />   ← your screen's JSX
 *   </AppShell>
 *
 * The sidebar collapses to icon-only on desktop (toggle button) and becomes a
 * full-width drawer on mobile (hamburger). Role-aware nav items are derived from
 * the current user's role — the server is always the security gate; the sidebar
 * only hides links the user cannot use.
 */

const ROLE_LABEL: Record<UserRole, string> = {
  STUDENT:     'Student',
  EXTERNAL:    'External',
  COORDINATOR: 'Coordinator',
  SUPER_ADMIN: 'Admin Staff',
};

const ROLE_COLOR: Record<UserRole, string> = {
  STUDENT:     '#498473',
  EXTERNAL:    '#2a6d9e',
  COORDINATOR: '#5a7b9a',
  SUPER_ADMIN: '#0B3754',
};

// ─────────────────────────────────────────────
//  AppShell
// ─────────────────────────────────────────────

interface AppShellProps {
  children: ReactNode;
  title?: string;
}

export default function AppShell({ children, title }: AppShellProps) {
  const { user, logout } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();
  const [collapsed, setCollapsed] = useState(() => window.innerWidth < 1100);
  const [drawerOpen, setDrawerOpen] = useState(false);
  const [isMobile, setIsMobile] = useState(window.innerWidth < 768);
  const drawerRef = useRef<HTMLDivElement>(null);

  // Responsive: track mobile breakpoint
  useEffect(() => {
    const mq = window.matchMedia('(max-width: 767px)');
    const handler = (e: MediaQueryListEvent) => {
      setIsMobile(e.matches);
      if (!e.matches) setDrawerOpen(false);
    };
    mq.addEventListener('change', handler);
    return () => mq.removeEventListener('change', handler);
  }, []);

  // Close drawer on route change
  useEffect(() => { setDrawerOpen(false); }, [location.pathname]);

  // Trap focus & close on outside click for mobile drawer
  useEffect(() => {
    if (!drawerOpen) return;
    const handler = (e: MouseEvent) => {
      if (drawerRef.current && !drawerRef.current.contains(e.target as Node)) {
        setDrawerOpen(false);
      }
    };
    document.addEventListener('mousedown', handler);
    return () => document.removeEventListener('mousedown', handler);
  }, [drawerOpen]);

  const visibleItems = user
    ? NAV_ITEMS.filter((item) => item.roles.includes(user.role))
    : [];

  const handleLogout = useCallback(async () => {
    await logout();
    navigate('/', { replace: true });
  }, [logout, navigate]);

  const sidebarContent = (isDrawer = false) => {
    let lastSection = '';
    return (
      <nav
        aria-label="Main navigation"
        style={{
          display: 'flex',
          flexDirection: 'column',
          height: '100%',
          overflowY: 'auto',
          overflowX: 'hidden',
          padding: '0 0 var(--sp-4)',
        }}
      >
        {/* Logo area */}
        <div style={{
          padding: (collapsed && !isDrawer) ? 'var(--sp-4) 0' : 'var(--sp-4) var(--sp-4)',
          display: 'flex',
          alignItems: 'center',
          gap: 10,
          justifyContent: (collapsed && !isDrawer) ? 'center' : 'flex-start',
          borderBottom: '1px solid rgba(255,255,255,0.10)',
          marginBottom: 'var(--sp-3)',
          minHeight: 64,
        }}>
          <div style={{
            width: 34, height: 34, borderRadius: 8,
            background: 'rgba(255,255,255,0.15)',
            display: 'flex', alignItems: 'center', justifyContent: 'center',
            flexShrink: 0,
            fontSize: 16, fontWeight: 700, fontFamily: 'var(--font-display)',
            color: '#fff',
          }}>B</div>
          {(!collapsed || isDrawer) && (
            <div style={{ overflow: 'hidden' }}>
              <div style={{
                font: '600 13px/1 var(--font-display)',
                color: '#fff',
                letterSpacing: '0.01em',
                whiteSpace: 'nowrap',
              }}>BUKC Sports</div>
              <div style={{
                font: '11px/1 var(--font-body)',
                color: 'rgba(255,255,255,0.55)',
                marginTop: 3,
                whiteSpace: 'nowrap',
              }}>Bahria University</div>
            </div>
          )}
        </div>

        {/* Nav items */}
        <div style={{ flex: 1, padding: '0 var(--sp-2)' }}>
          {visibleItems.map((item) => {
            const isFirst = item.section && item.section !== lastSection;
            if (isFirst) lastSection = item.section!;

            return (
              <div key={item.to}>
                {isFirst && (!collapsed || isDrawer) && (
                  <div style={{
                    font: '500 10px/1 var(--font-body)',
                    color: 'rgba(255,255,255,0.38)',
                    letterSpacing: '0.08em',
                    textTransform: 'uppercase',
                    padding: 'var(--sp-3) var(--sp-2) var(--sp-1)',
                  }}>
                    {item.section}
                  </div>
                )}
                {isFirst && (collapsed && !isDrawer) && (
                  <div style={{
                    height: 1,
                    background: 'rgba(255,255,255,0.08)',
                    margin: 'var(--sp-2) var(--sp-2)',
                  }} />
                )}
                <NavLink
                  to={item.to}
                  title={collapsed && !isDrawer ? item.label : undefined}
                  style={({ isActive }) => ({
                    display: 'flex',
                    alignItems: 'center',
                    gap: 10,
                    padding: (collapsed && !isDrawer)
                      ? 'var(--sp-3) 0'
                      : 'var(--sp-2) var(--sp-3)',
                    borderRadius: 'var(--radius)',
                    margin: '1px 0',
                    color: isActive ? '#fff' : 'rgba(255,255,255,0.65)',
                    background: isActive ? 'rgba(255,255,255,0.14)' : 'transparent',
                    textDecoration: 'none',
                    fontSize: 13.5,
                    fontWeight: isActive ? 600 : 400,
                    transition: 'background var(--t-fast) var(--ease), color var(--t-fast) var(--ease)',
                    justifyContent: (collapsed && !isDrawer) ? 'center' : 'flex-start',
                    position: 'relative',
                    whiteSpace: 'nowrap',
                    overflow: 'hidden',
                  })}
                >
                  {({ isActive }) => (
                    <>
                      <span style={{
                        flexShrink: 0,
                        color: isActive ? '#fff' : 'rgba(255,255,255,0.55)',
                        display: 'flex',
                        transition: 'color var(--t-fast) var(--ease)',
                      }}>
                        {item.icon}
                      </span>
                      {(!collapsed || isDrawer) && (
                        <span style={{ overflow: 'hidden', textOverflow: 'ellipsis' }}>
                          {item.label}
                        </span>
                      )}
                      {isActive && (
                        <span style={{
                          position: 'absolute',
                          left: 0, top: '50%', transform: 'translateY(-50%)',
                          width: 3, height: '60%',
                          background: 'var(--teal-500)',
                          borderRadius: '0 2px 2px 0',
                        }} />
                      )}
                    </>
                  )}
                </NavLink>
              </div>
            );
          })}
        </div>

        {/* User + logout */}
        {user && (
          <div style={{
            padding: 'var(--sp-3) var(--sp-3)',
            borderTop: '1px solid rgba(255,255,255,0.08)',
            marginTop: 'auto',
            display: 'flex',
            alignItems: 'center',
            gap: 10,
          }}>
            <div style={{
              width: 32, height: 32, borderRadius: '50%',
              background: ROLE_COLOR[user.role],
              display: 'flex', alignItems: 'center', justifyContent: 'center',
              flexShrink: 0,
              font: '600 12px var(--font-display)',
              color: '#fff',
              border: '2px solid rgba(255,255,255,0.2)',
            }}>
              {user.fullName.charAt(0).toUpperCase()}
            </div>
            {(!collapsed || isDrawer) && (
              <>
                <div style={{ flex: 1, minWidth: 0 }}>
                  <div style={{
                    font: '500 12.5px var(--font-body)',
                    color: '#fff',
                    overflow: 'hidden',
                    textOverflow: 'ellipsis',
                    whiteSpace: 'nowrap',
                  }}>
                    {user.fullName}
                  </div>
                  <div style={{
                    font: '11px var(--font-body)',
                    color: 'rgba(255,255,255,0.45)',
                    marginTop: 1,
                  }}>
                    {ROLE_LABEL[user.role]}
                  </div>
                </div>
                <button
                  onClick={() => { void handleLogout(); }}
                  title="Sign out"
                  aria-label="Sign out"
                  style={{
                    background: 'none', border: 'none', cursor: 'pointer',
                    color: 'rgba(255,255,255,0.45)', padding: 4, borderRadius: 4,
                    display: 'flex', transition: 'color var(--t-fast)',
                    flexShrink: 0,
                  }}
                  onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = 'rgba(255,255,255,0.9)'; }}
                  onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = 'rgba(255,255,255,0.45)'; }}
                >
                  {ic.logout}
                </button>
              </>
            )}
          </div>
        )}
      </nav>
    );
  };

  return (
    <div style={{ display: 'flex', height: '100%', overflow: 'hidden' }}>

      {/* ── Desktop sidebar ── */}
      {!isMobile && (
        <aside style={{
          width: collapsed ? 'var(--sidebar-w-col)' : 'var(--sidebar-w)',
          minWidth: collapsed ? 'var(--sidebar-w-col)' : 'var(--sidebar-w)',
          background: 'var(--navy)',
          display: 'flex',
          flexDirection: 'column',
          position: 'relative',
          zIndex: 30,
          transition: 'width var(--t-base) var(--ease), min-width var(--t-base) var(--ease)',
          overflowX: 'hidden',
        }}>
          {sidebarContent(false)}

          {/* Collapse toggle */}
          <button
            onClick={() => setCollapsed((c) => !c)}
            aria-label={collapsed ? 'Expand sidebar' : 'Collapse sidebar'}
            style={{
              position: 'absolute',
              top: 20,
              right: collapsed ? '50%' : 12,
              transform: collapsed ? 'translateX(50%)' : 'none',
              width: 24, height: 24,
              borderRadius: '50%',
              background: 'rgba(255,255,255,0.12)',
              border: '1px solid rgba(255,255,255,0.18)',
              color: 'rgba(255,255,255,0.7)',
              cursor: 'pointer',
              display: 'flex', alignItems: 'center', justifyContent: 'center',
              transition: 'all var(--t-base) var(--ease)',
              zIndex: 10,
              padding: 0,
            }}
          >
            <span style={{
              display: 'flex',
              transform: collapsed ? 'rotate(180deg)' : 'none',
              transition: 'transform var(--t-base) var(--ease)',
            }}>
              {ic.chevronLeft}
            </span>
          </button>
        </aside>
      )}

      {/* ── Mobile drawer overlay ── */}
      {isMobile && drawerOpen && (
        <div
          style={{
            position: 'fixed', inset: 0, zIndex: 40,
            background: 'rgba(11,55,84,0.5)',
            backdropFilter: 'blur(2px)',
            animation: 'fadeIn var(--t-fast) var(--ease)',
          }}
          aria-hidden
        />
      )}

      {/* ── Mobile drawer ── */}
      {isMobile && (
        <aside
          ref={drawerRef}
          style={{
            position: 'fixed',
            top: 0, left: 0, bottom: 0,
            width: 260,
            background: 'var(--navy)',
            zIndex: 50,
            transform: drawerOpen ? 'translateX(0)' : 'translateX(-100%)',
            transition: 'transform var(--t-base) var(--ease)',
            display: 'flex',
            flexDirection: 'column',
            overflowY: 'auto',
          }}
        >
          <button
            onClick={() => setDrawerOpen(false)}
            aria-label="Close menu"
            style={{
              position: 'absolute', top: 16, right: 12,
              background: 'rgba(255,255,255,0.1)', border: 'none',
              borderRadius: 6, padding: 6, cursor: 'pointer',
              color: 'rgba(255,255,255,0.8)', display: 'flex',
            }}
          >
            {ic.close}
          </button>
          {sidebarContent(true)}
        </aside>
      )}

      {/* ── Main area (topbar + content) ── */}
      <div style={{ flex: 1, display: 'flex', flexDirection: 'column', overflow: 'hidden', minWidth: 0 }}>

        {/* Topbar */}
        <header style={{
          height: 56,
          background: 'var(--white)',
          borderBottom: '1px solid var(--line-light)',
          display: 'flex',
          alignItems: 'center',
          padding: '0 var(--sp-5)',
          gap: 'var(--sp-4)',
          flexShrink: 0,
          boxShadow: 'var(--shadow-xs)',
          zIndex: 20,
        }}>
          {/* Mobile hamburger */}
          {isMobile && (
            <button
              onClick={() => setDrawerOpen(true)}
              aria-label="Open menu"
              style={{
                background: 'none', border: 'none', cursor: 'pointer',
                color: 'var(--navy)', display: 'flex', padding: 4, borderRadius: 4,
                marginRight: 4,
              }}
            >
              {ic.menu}
            </button>
          )}

          {/* Page title */}
          {title && (
            <h1 style={{
              font: '600 16px/1 var(--font-display)',
              color: 'var(--navy)',
              margin: 0,
              flex: 1,
            }}>
              {title}
            </h1>
          )}
          {!title && <div style={{ flex: 1 }} />}

          {/* Right side: user pill */}
          {user && (
            <div style={{
              display: 'flex',
              alignItems: 'center',
              gap: 8,
              padding: '4px 10px 4px 6px',
              borderRadius: 20,
              background: 'var(--bg)',
              border: '1px solid var(--line)',
            }}>
              <div style={{
                width: 26, height: 26, borderRadius: '50%',
                background: ROLE_COLOR[user.role],
                display: 'flex', alignItems: 'center', justifyContent: 'center',
                font: '600 11px var(--font-display)',
                color: '#fff',
                flexShrink: 0,
              }}>
                {user.fullName.charAt(0).toUpperCase()}
              </div>
              <div>
                <div style={{ font: '500 12px var(--font-body)', color: 'var(--navy)', lineHeight: 1 }}>
                  {user.fullName.split(' ')[0]}
                </div>
                <div style={{ font: '10.5px var(--font-body)', color: 'var(--ink-muted)', marginTop: 1 }}>
                  {ROLE_LABEL[user.role]}
                </div>
              </div>
            </div>
          )}
        </header>

        {/* Scrollable content */}
        <main
          id="main-content"
          style={{
            flex: 1,

/** Primary action button */
export function Btn({
  children, variant = 'primary', size = 'md', loading: isLoading, ...props
}: React.ButtonHTMLAttributes<HTMLButtonElement> & {
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  loading?: boolean;
}) {
  const colors: Record<string, string> = {
    primary:   'background:var(--teal);color:#fff;border:1px solid var(--teal-700);',
    secondary: 'background:var(--white);color:var(--navy);border:1px solid var(--line);',
    danger:    'background:var(--danger-bg);color:var(--danger);border:1px solid #f3caca;',
    ghost:     'background:transparent;color:var(--teal);border:1px solid transparent;',
  };
  const sizes: Record<string, string> = {
    sm: 'padding:5px 12px;font-size:12.5px;',
    md: 'padding:8px 18px;font-size:13.5px;',
    lg: 'padding:11px 24px;font-size:15px;',
  };
  return (
    <button
      {...props}
      disabled={props.disabled || isLoading}
      style={{
        ...Object.fromEntries(
          (colors[variant] + sizes[size])
            .split(';').filter(Boolean)
            .map((p) => p.split(':').map((v) => v.trim()))
        ),
        fontFamily: 'var(--font-body)',
        fontWeight: 600,
        borderRadius: 'var(--radius)',
        cursor: props.disabled || isLoading ? 'not-allowed' : 'pointer',
        opacity: props.disabled || isLoading ? 0.6 : 1,
        display: 'inline-flex',
        alignItems: 'center',
        gap: 6,
        transition: 'opacity var(--t-fast), box-shadow var(--t-fast)',
        whiteSpace: 'nowrap',
        ...props.style,
      }}
    >
      {isLoading ? '…' : children}
    </button>
  );
}
